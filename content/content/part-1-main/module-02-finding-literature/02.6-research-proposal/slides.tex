\begin{frame}{The Heilmeier Catechism}
  \textit{7 questions every proposal must answer (from DARPA Director George Heilmeier):}
  \vspace{0.3cm}
  \begin{enumerate}
    \item What are you trying to do? (plain English)
    \item How is it done today, and what are the limits?
    \item What is new in your approach?
    \item Who cares? What difference does success make?
    \item What are the risks and the payoffs?
    \item How much will it cost and how long will it take?
    \item What are the mid-term and final ``exams'' for success?
  \end{enumerate}
\end{frame}

\begin{frame}{The Logic Model}
  Reviewers trace a direct line from money to world impact.
  \vspace{0.5cm}
  \begin{center}
  \textbf{Inputs} $\rightarrow$ \textbf{Activities} $\rightarrow$ \textbf{Outputs} $\rightarrow$ \textbf{Outcomes} $\rightarrow$ \textbf{Impact}
  \end{center}
  \vspace{0.3cm}
  \begin{exampleblock}{Example}
    Budget \$150k $\rightarrow$ Run clinical trial $\rightarrow$ Dataset + paper $\rightarrow$ New treatment guideline $\rightarrow$ Fewer cancer deaths
  \end{exampleblock}
\end{frame}

\begin{frame}{SMART Objectives}
  \begin{alertblock}{Vague (Rejected)}
    ``To investigate the impact of X on Y.''
  \end{alertblock}
  \begin{exampleblock}{SMART (Fundable)}
    ``To measure the effect of \textbf{[intervention X]} on \textbf{[outcome Y]} in \textbf{[population Z]}, using \textbf{[methodology M]}, within \textbf{18 months}, targeting an effect size of \textbf{d = 0.5}.''
  \end{exampleblock}
\end{frame}

\begin{frame}{The 5-Section Structure}
  \begin{enumerate}
    \item \textbf{Significance} — Why does this problem matter? (Heilmeier Q1, Q2, Q4)
    \item \textbf{Innovation} — What is genuinely new? (Heilmeier Q3)
    \item \textbf{Approach} — How will you do it? Timeline + milestones (Heilmeier Q6, Q7)
    \item \textbf{Team} — Why are you the right person?
    \item \textbf{Budget Justification} — Every line item explained.
  \end{enumerate}
\end{frame}

\begin{frame}[standout]
