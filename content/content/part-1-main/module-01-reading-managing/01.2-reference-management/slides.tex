\begin{frame}{The Problem with Manual Citations}
  \begin{alertblock}{Manual Formatting is a Trap}
    Typing citations manually means if you submit to a new journal with a different style (e.g., APA $\rightarrow$ IEEE), you must reformat every single comma, italic, and initial by hand.
  \end{alertblock}
  \begin{exampleblock}{The Solution}
    A reference manager separates the \textbf{metadata} (the data about the paper) from the \textbf{presentation} (the final formatted text).
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Under the Hood: BibTeX}
  BibTeX is the universal plain-text standard for academic metadata.
  
\begin{verbatim}
@article{vaswani2017attention,
  title={Attention is all you need},
  author={Vaswani, Ashish and Shazeer, Noam},
  journal={Advances in neural information processing systems},
  volume={30},
  year={2017}
}
\end{verbatim}

  \textit{The text \texttt{vaswani2017attention} is the \textbf{Citation Key}. You use this ID to insert the reference into your document.}
\end{frame}
