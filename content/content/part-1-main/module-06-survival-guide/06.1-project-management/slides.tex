\begin{frame}{What You'll Learn}
  \begin{enumerate}
    \item \textbf{Apply} agile principles to academic research.
    \item \textbf{Deconstruct} a thesis using a Kanban board.
    \item \textbf{Manage} your advisor with structured agendas.
    \item \textbf{Implement} a Personal Knowledge Management (PKM) system.
  \end{enumerate}
\end{frame}

\begin{frame}{The Academic Kanban Board}
  \begin{center}
  \begin{tabular}{|p{2.5cm}|p{2.5cm}|p{2.5cm}|p{2.5cm}|p{2.5cm}|}
  \hline
  \textbf{Backlog} & \textbf{To Do (Sprint)} & \textbf{Doing} & \textbf{Waiting} & \textbf{Done} \\
  \hline
  \textit{Ideas, future experiments.} & \textit{Tasks for the next 2 weeks.} & \textit{What you are doing TODAY. Max 2 items.} & \textit{Waiting on peer review or code runs.} & \textit{Finished work.} \\
  \hline
  \end{tabular}
  \end{center}
  \vspace{0.5cm}
  \textit{Visualizing work prevents cognitive overload. You no longer worry about the ``mountain,'' only the ``Doing'' column.}
\end{frame}

\begin{frame}{Managing Up: The Meeting Agenda}
  You are the CEO of your project. Your advisor is the Board of Directors.
  \vspace{0.3cm}
  \begin{exampleblock}{The 4-Part Update Agenda}
    \begin{enumerate}
      \item \textbf{Progress:} What I accomplished since our last meeting.
      \item \textbf{Roadblock:} The specific issue I am facing.
      \item \textbf{Proposed Solution:} How I think we should fix it.
      \item \textbf{Action Needed:} Exactly what I need from you today (e.g., ``read this draft,'' ``approve this budget'').
    \end{enumerate}
  \end{exampleblock}
\end{frame}

\begin{frame}{Personal Knowledge Management (PKM)}
  \begin{itemize}
    \item You cannot hold 4 years of reading in your working memory.
    \item \textbf{Zettelkasten Method:} Take ``atomic'' notes (one concept per note).
    \item Link notes together (using tools like Obsidian, Notion, or Roam).
    \item \textbf{The Result:} When writing your thesis, you assemble pre-written notes instead of starting from a blank page.
  \end{itemize}
\end{frame}
