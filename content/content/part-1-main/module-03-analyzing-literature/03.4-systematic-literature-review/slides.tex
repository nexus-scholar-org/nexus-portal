\begin{frame}{Narrative vs. Systematic}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Narrative Review}
        \begin{itemize}
          \item Author-driven selection
          \item Subjective inclusion
          \item Not reproducible
          \item Susceptible to cherry-picking
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Systematic Review}
        \begin{itemize}
          \item Protocol-driven
          \item Pre-defined criteria
          \item Fully reproducible
          \item Highest level of evidence
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
