\begin{frame}{What You'll Learn}
  \begin{enumerate}
    \item \textbf{Differentiate} ethical vs. unethical LLM use.
    \item \textbf{Apply} LLMs as editors, not ghostwriters.
    \item \textbf{Evaluate} the risks of AI hallucinations.
    \item \textbf{Draft} a transparency statement.
  \end{enumerate}
\end{frame}

\begin{frame}{Copilot, Not Autopilot}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Unethical (Ghostwriter)}
        \begin{itemize}
          \item ``Write my introduction.''
          \item ``Find papers to cite.''
          \item Generates ideas/data.
          \item Academic misconduct.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Ethical (Copilot)}
        \begin{itemize}
          \item ``Check my grammar.''
          \item ``Improve this transition.''
          \item Refines your existing text.
          \item Highly encouraged.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The Danger of Hallucinations}
  \begin{center}
    \Large \textbf{LLMs are statistical prediction engines, not databases.}
  \end{center}
  \vspace{0.5cm}
  \begin{itemize}
    \item They will invent fake authors.
    \item They will invent fake paper titles.
    \item They will invent fake DOIs that look mathematically correct.
  \end{itemize}
  \vspace{0.5cm}
  \begin{alertblock}{The Rule of Accountability}
    You are 100\% legally and morally responsible for every word and citation in your paper. ``The AI made a mistake'' is not a defense.
  \end{alertblock}
\end{frame}

\begin{frame}{Prompting for Structure}
  \textbf{Bad Prompt:} \\
  \textit{``Write a conclusion for my paper.''}
  
  \vspace{0.5cm}
  
  \textbf{Good Prompt:} \\
  \textit{``Here is my draft conclusion. Does it follow the IMRaD format? Point out any sentences longer than 25 words and suggest edits for clarity.''}
\end{frame}

\begin{frame}{Transparency \& Disclosure}
  Most major publishers (Nature, IEEE, Elsevier) require disclosure.
  \vspace{0.5cm}
  \begin{exampleblock}{Standard Disclosure Template}
    ``During the preparation of this work, the author(s) used [Tool Name] in order to [Reason, e.g., improve readability]. After using this tool/service, the author(s) reviewed and edited the content as needed and take(s) full responsibility for the content of the publication.''
  \end{exampleblock}
\end{frame}

\begin{frame}[standout]
  Use AI to sharpen your voice,\\
  not to replace it.
\end{frame}
