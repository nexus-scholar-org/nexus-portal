\begin{frame}{The IMRaD Blueprint}
  \begin{table}
  \begin{tabular}{lll}
  \toprule
  \textbf{Section} & \textbf{Question It Answers} & \textbf{Write Order} \\
  \midrule
  Introduction & Why should anyone care? & 4th (last) \\
  Methods & How did you do it? & 1st (easiest) \\
  Results & What did you find? & 2nd \\
  Discussion & What does it mean? & 3rd \\
  \bottomrule
  \end{tabular}
  \end{table}
  \vspace{0.3cm}
  \textit{Never start with the Introduction. Start with Methods --- you already know what you did.}
\end{frame}

\begin{frame}{Common Mistakes}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Don't}
        \begin{itemize}
          \item ``This proves that X.''
          \item ``The experiment was conducted by us.''
          \item Sentences longer than 25 words.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Do}
        \begin{itemize}
          \item ``These results suggest that X.''
          \item ``We conducted the experiment.''
          \item Short, clear sentences with strong verbs.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Your Turn: Exercise}
  \begin{exampleblock}{Exercise: The Reverse Outline}

  \end{exampleblock}
\end{frame}
