\begin{frame}{What You'll Learn}
  \begin{enumerate}
    \item \textbf{Construct} a point-by-point response letter.
    \item \textbf{Apply} professional strategies for common reviewer scenarios.
    \item \textbf{Distinguish} defensive vs. professional responses.
    \item \textbf{Reframe} rejection as redirection.
  \end{enumerate}
\end{frame}

\begin{frame}{The Point-by-Point Format}
  For \textbf{every} reviewer comment:
  \vspace{0.3cm}
  \begin{enumerate}
    \item \textbf{Quote} the original comment verbatim, in italics.
    \item \textbf{Respond} directly, respectfully, and thoroughly.
    \item \textbf{Show the change} --- exact new text with page \& line numbers.
  \end{enumerate}
  \vspace{0.3cm}
  \begin{alertblock}{Golden Rule}
    \textbf{Never be defensive.} Even when the reviewer is wrong, thank them for raising the point.
  \end{alertblock}
\end{frame}

\begin{frame}{Common Scenarios}
  \begin{table}
  \footnotesize
  \begin{tabular}{lp{6cm}}
  \toprule
  \textbf{Reviewer Says} & \textbf{You Respond} \\
  \midrule
  ``You need more data.'' & Add what you can; explain constraints. \\
  ``I disagree with your interpretation.'' & Acknowledge the alternative view in Discussion. \\
  ``This was done by Smith (2019).'' & Differentiate clearly; cite Smith. \\
  \bottomrule
  \end{tabular}
  \end{table}
\end{frame}

\begin{frame}{Rejection = Redirection}
  \begin{itemize}
    \item Read the reviews for honest feedback.
    \item Revise your paper accordingly.
    \item Identify a more appropriate journal.
    \item Resubmit. This is the normal cycle.
  \end{itemize}
  \vspace{0.3cm}
  \textit{The most-cited papers in many fields were rejected from their first-choice journal.}
\end{frame}
