\begin{frame}{The Name Problem}
  \begin{center}
    \textit{Are you J. Smith, Jon Smith, or Jonathan A. Smith?}
  \end{center}
  \vspace{0.3cm}
  \begin{alertblock}{The Solution: ORCID}
    The Open Researcher and Contributor ID is a unique 16-digit number that acts as your academic social security number. It links all your papers, regardless of how a journal formats your name.
  \end{alertblock}
  \begin{exampleblock}{Google Scholar}
    The most viewed academic profile in the world. Claim it so you control your publication list and citations.
  \end{exampleblock}
\end{frame}

\begin{frame}{The Minimal Academic Website}
  You do not need a flashy portfolio. You need a digital business card.
  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{Bio:} Photo + 2 sentences on what you study.
    \item \textbf{Publications:} A list with direct links to the PDFs.
    \item \textbf{CV:} A downloadable PDF.
    \item \textbf{Contact:} How to reach you.
  \end{itemize}
  \vspace{0.3cm}
  \textit{Tools: GitHub Pages, Notion, or basic WordPress/Squarespace.}
\end{frame}

\begin{frame}{Academic Social Media}
  \begin{columns}[T]

      \textbf{Twitter/X}
      \begin{itemize}

        \item Great for early paper discovery and jobs.
        \item Be a curator, not just a broadcaster.
      \end{itemize}
    \end{column}
    \begin{column}{0.33\textwidth}
      \textbf{LinkedIn}
      \begin{itemize}
        \item Essential for industry research roles.
        \item Postdoc networking.
      \end{itemize}
    \end{column}
    \begin{column}{0.33\textwidth}
      \textbf{ResearchGate}
      \begin{itemize}
        \item Caution: For-profit.
        \item Often violates copyright by hosting PDFs.
        \item Can appear spammy.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
