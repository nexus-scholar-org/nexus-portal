\begin{frame}{What You'll Learn}
  \begin{enumerate}
    \item \textbf{Design} a poster using the ``Billboard'' principle.
    \item \textbf{Structure} a 10-minute conference talk.
    \item \textbf{Deliver} an effective elevator pitch.
    \item \textbf{Evaluate} and avoid common presentation mistakes.
  \end{enumerate}
\end{frame}

\begin{frame}{The Real Purpose of a Conference}
  \begin{alertblock}{Misconception}
    ``I must explain every detail of my 3-year project so they understand exactly what I did.''
  \end{alertblock}
  \vspace{0.5cm}
  \begin{exampleblock}{Reality}
    A conference presentation is a \textbf{trailer for your movie}. Your goal is to advertise your work, network, and convince people to read your paper.
  \end{exampleblock}
\end{frame}

\begin{frame}{The Modern Poster: The ``Billboard''}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Traditional (Fails)}
      \begin{itemize}
        \item Wall of 10-point text.
        \item Unreadable from 5 feet away.
        \item Acts as a standalone document.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Modern (Succeeds)}
      \begin{itemize}
        \item Massive central takeaway sentence.
        \item Highly visual (charts over text).
        \item QR code to the full paper.
        \item Acts as a conversation prop.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The 10-Minute Talk Structure}
  \begin{table}
  \begin{tabular}{llp{7cm}}
  \toprule
  \textbf{Time} & \textbf{Phase} & \textbf{Focus} \\
  \midrule
  Min 1 & The Hook & Why does this matter? \\
  Min 2--3 & The Context & What is the specific gap in the literature? \\
  Min 4--8 & The Contribution & Your key findings (skip standard methods). \\
  Min 9--10 & The Takeaway & The one thing they should remember tomorrow. \\
  \bottomrule
  \end{tabular}
  \end{table}
\end{frame}

\begin{frame}{The Elevator Pitch}
  \textbf{Goal: 30 seconds in the coffee line.}
  \vspace{0.3cm}
  \begin{enumerate}
    \item \textbf{Who:} ``I'm [Name], a PhD student at [University].''
    \item \textbf{Problem \& Impact:} ``I study [Problem] because [Impact].''
    \item \textbf{Finding:} ``Recently, we found [Key Finding], which means [Implication].''
  \end{enumerate}
\end{frame}

\begin{frame}[standout]
  Your research deserves to be seen.\\
  Stop hiding it behind walls of text.
\end{frame}
